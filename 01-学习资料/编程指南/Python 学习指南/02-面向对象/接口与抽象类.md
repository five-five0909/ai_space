# 接口与抽象类 (ABC)

Python 使用 `abc` 模块实现抽象基类（Abstract Base Class）。

## 基础语法

```python
from abc import ABC, abstractmethod

class Shape(ABC):
    """抽象基类"""

    @abstractmethod
    def area(self):
        """计算面积（子类必须实现）"""
        pass

    @abstractmethod
    def perimeter(self):
        """计算周长（子类必须实现）"""
        pass

class Rectangle(Shape):
    def __init__(self, width, height):
        self.width = width
        self.height = height

    def area(self):
        return self.width * self.height

    def perimeter(self):
        return 2 * (self.width + self.height)

# 不能实例化抽象类
# shape = Shape()  # TypeError!
```

> **提示**：`@abstractmethod` 装饰的方法，子类**必须**实现，否则无法实例化。

## LeetCode 实战场景

### 1. 定义算法接口

```python
from abc import ABC, abstractmethod
from typing import List

class SortingAlgorithm(ABC):
    """排序算法接口"""

    @abstractmethod
    def sort(self, nums: List[int]) -> List[int]:
        pass

class QuickSort(SortingAlgorithm):
    def sort(self, nums: List[int]) -> List[int]:
        if len(nums) <= 1:
            return nums
        pivot = nums[len(nums) // 2]
        left = [x for x in nums if x < pivot]
        middle = [x for x in nums if x == pivot]
        right = [x for x in nums if x > pivot]
        return self.sort(left) + middle + self.sort(right)

class MergeSort(SortingAlgorithm):
    def sort(self, nums: List[int]) -> List[int]:
        if len(nums) <= 1:
            return nums
        mid = len(nums) // 2
        left = self.sort(nums[:mid])
        right = self.sort(nums[mid:])
        return self._merge(left, right)

    def _merge(self, left: List[int], right: List[int]) -> List[int]:
        result = []
        i = j = 0
        while i < len(left) and j < len(right):
            if left[i] <= right[j]:
                result.append(left[i])
                i += 1
            else:
                result.append(right[j])
                j += 1
        result.extend(left[i:])
        result.extend(right[j:])
        return result
```

### 2. 定义图遍历接口

```python
from abc import ABC, abstractmethod
from collections import deque
from typing import List, Dict, Any

class GraphTraversal(ABC):
    """图遍历算法接口"""

    @abstractmethod
    def traverse(self, graph: Dict[Any, List[Any]], start: Any) -> List[Any]:
        pass

class BFS(GraphTraversal):
    """广度优先搜索"""
    def traverse(self, graph: Dict[Any, List[Any]], start: Any) -> List[Any]:
        visited = set()
        queue = deque([start])
        visited.add(start)
        result = []

        while queue:
            node = queue.popleft()
            result.append(node)
            for neighbor in graph.get(node, []):
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append(neighbor)
        return result

class DFS(GraphTraversal):
    """深度优先搜索"""
    def traverse(self, graph: Dict[Any, List[Any]], start: Any) -> List[Any]:
        visited = set()
        result = []

        def dfs(node):
            visited.add(node)
            result.append(node)
            for neighbor in graph.get(node, []):
                if neighbor not in visited:
                    dfs(neighbor)

        dfs(start)
        return result
```

## 科研实战场景

### 1. 定义 Dataset 接口

```python
from abc import ABC, abstractmethod
from typing import Tuple
import torch

class BaseDataset(ABC):
    """数据集抽象基类"""

    @abstractmethod
    def __len__(self) -> int:
        pass

    @abstractmethod
    def __getitem__(self, index: int) -> Tuple[torch.Tensor, torch.Tensor]:
        pass

    @property
    @abstractmethod
    def num_classes(self) -> int:
        pass

# 实现具体数据集
class CIFAR10Dataset(BaseDataset):
    def __init__(self, root: str, train: bool = True):
        from torchvision.datasets import CIFAR10
        self.dataset = CIFAR10(root=root, train=train, download=True)

    def __len__(self) -> int:
        return len(self.dataset)

    def __getitem__(self, index: int):
        return self.dataset[index]

    @property
    def num_classes(self) -> int:
        return 10
```

### 2. 定义 Model 接口

```python
from abc import ABC, abstractmethod
from typing import Dict, Any
import torch
import torch.nn as nn

class BaseModel(ABC):
    """模型抽象基类"""

    def __init__(self, config: Dict[str, Any]):
        self.config = config
        self.model = self._build_model()

    @abstractmethod
    def _build_model(self) -> nn.Module:
        pass

    @abstractmethod
    def forward(self, batch: Dict[str, torch.Tensor]) -> Dict[str, torch.Tensor]:
        pass

    def to(self, device):
        self.model.to(device)
        return self

    def save(self, path: str):
        torch.save(self.model.state_dict(), path)

# 实现具体模型
class ResNetClassifier(BaseModel):
    def _build_model(self) -> nn.Module:
        from torchvision.models import resnet50, ResNet50_Weights
        model = resnet50(weights=ResNet50_Weights.IMAGENET1K_V1)
        model.fc = nn.Linear(model.fc.in_features, self.config['num_classes'])
        return model

    def forward(self, batch: Dict[str, torch.Tensor]) -> Dict[str, torch.Tensor]:
        x = batch['image']
        logits = self.model(x)
        return {'logits': logits, 'preds': logits.argmax(dim=1)}
```

### 3. 定义训练器接口

```python
from abc import ABC, abstractmethod
from typing import Dict, List
import torch
import torch.nn as nn

class Trainer(ABC):
    """训练器抽象基类"""

    def __init__(self, model: nn.Module, config: Dict[str, Any]):
        self.model = model
        self.config = config
        self.history: List[Dict[str, float]] = []

    @abstractmethod
    def train_step(self, batch) -> Dict[str, torch.Tensor]:
        pass

    @abstractmethod
    def val_step(self, batch) -> Dict[str, torch.Tensor]:
        pass

    def fit(self, train_loader, val_loader, epochs: int):
        for epoch in range(epochs):
            train_logs = self._train_one_epoch(train_loader)
            val_logs = self._validate_one_epoch(val_loader)
            epoch_logs = {'epoch': epoch, **train_logs, **val_logs}
            self.history.append(epoch_logs)
            print(f"Epoch {epoch}: {epoch_logs}")
            if self._should_stop():
                print("触发早停")
                break
        return self.history

    def _train_one_epoch(self, loader) -> Dict[str, float]:
        self.model.train()
        total_loss = 0
        for batch in loader:
            logs = self.train_step(batch)
            total_loss += logs.get('loss', 0)
        return {'train_loss': total_loss / len(loader)}

    def _validate_one_epoch(self, loader) -> Dict[str, float]:
        self.model.eval()
        total_loss = 0
        with torch.no_grad():
            for batch in loader:
                logs = self.val_step(batch)
                total_loss += logs.get('loss', 0)
        return {'val_loss': total_loss / len(loader)}

    def _should_stop(self) -> bool:
        if len(self.history) < self.config.get('patience', 5):
            return False
        current = self.history[-1]['val_loss']
        best = min(h['val_loss'] for h in self.history[:-1])
        return current >= best

# 实现具体训练器
class ClassificationTrainer(Trainer):
    def __init__(self, model, config):
        super().__init__(model, config)
        self.optimizer = torch.optim.Adam(
            model.parameters(),
            lr=config.get('learning_rate', 1e-4)
        )
        self.criterion = nn.CrossEntropyLoss()

    def train_step(self, batch):
        self.optimizer.zero_grad()
        images, labels = batch
        outputs = self.model(images)
        loss = self.criterion(outputs['logits'], labels)
        loss.backward()
        self.optimizer.step()
        return {'loss': loss.item()}

    def val_step(self, batch):
        images, labels = batch
        outputs = self.model(images)
        loss = self.criterion(outputs['logits'], labels)
        accuracy = (outputs['preds'] == labels).float().mean()
        return {'loss': loss.item(), 'accuracy': accuracy.item()}
```

## 性能提示

> **抽象类开销**：
> - 抽象类本身几乎没有运行时开销
> - `@abstractmethod` 只在实例化时检查
> - 接口调用与普通方法调用性能相同

> **设计建议**：
> - 科研项目中使用 ABC 可以确保不同数据集/模型/训练器有统一接口
> - 便于测试和替换组件
> - 但简单脚本不需要过度设计
