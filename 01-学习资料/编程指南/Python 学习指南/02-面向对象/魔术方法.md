# 魔术方法 (Magic Methods)

魔术方法（双下划线方法）让自定义类可以模拟 Python 内置类型的行为。

## 常用魔术方法

### 字符串表示

```python
class Person:
    def __init__(self, name, age):
        self.name = name
        self.age = age

    def __str__(self):
        """用户友好的字符串表示（print() 使用）"""
        return f"{self.name}, {self.age}岁"

    def __repr__(self):
        """开发者友好的表示（交互式解释器使用）"""
        return f"Person(name={self.name!r}, age={self.age})"

p = Person("Alice", 25)
print(p)           # Alice, 25 岁
print(repr(p))     # Person(name='Alice', age=25)
```

> **提示**：`__repr__` 的目标是**无歧义**，最好能用于重建对象；`__str__` 的目标是**可读性**。

### 容器行为

```python
class MyList:
    def __init__(self, items):
        self._items = items

    def __len__(self):
        return len(self._items)

    def __getitem__(self, index):
        return self._items[index]

    def __setitem__(self, index, value):
        self._items[index] = value

    def __iter__(self):
        return iter(self._items)

    def __contains__(self, item):
        return item in self._items
```

### 数值运算

```python
class Vector:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __add__(self, other):
        return Vector(self.x + other.x, self.y + other.y)

    def __sub__(self, other):
        return Vector(self.x - other.x, self.y - other.y)

    def __mul__(self, scalar):
        return Vector(self.x * scalar, self.y * scalar)

    def __eq__(self, other):
        return self.x == other.x and self.y == other.y

    def __repr__(self):
        return f"Vector({self.x}, {self.y})"
```

### 上下文管理器

```python
class FileManager:
    def __init__(self, filename, mode):
        self.filename = filename
        self.mode = mode

    def __enter__(self):
        self.file = open(self.filename, self.mode)
        return self.file

    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.file:
            self.file.close()
        return False

# 使用
with FileManager("test.txt", "w") as f:
    f.write("Hello")
```

### 1. 自定义迭代器（LC 173: 二叉搜索树迭代器）

```python
class BSTIterator:
    """LC 173: 二叉搜索树迭代器"""

    def __init__(self, root):
        self.stack = []
        self._push_left(root)

    def _push_left(self, node):
        while node:
            self.stack.append(node)
            node = node.left

    def __next__(self) -> int:
        if not self.has_next():
            raise StopIteration
        node = self.stack.pop()
        if node.right:
            self._push_left(node.right)
        return node.val

    def has_next(self) -> bool:
        return len(self.stack) > 0

    def __iter__(self):
        return self
```

import heapq

class HeapItem:
    """可比较的堆元素（用于优先队列）"""

    def __init__(self, priority, value):

        self.value = value

    def __lt__(self, other):
        return self.priority < other.priority

    def __repr__(self):
        return f"HeapItem({self.priority}, {self.value!r})"

### 3. 可哈希的自定义对象

```python
class Point:
    """二维点，可哈希，可用作字典 key 或 set 元素"""

    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __eq__(self, other):
        if not isinstance(other, Point):
            return False
        return self.x == other.x and self.y == other.y

    def __hash__(self):
        return hash((self.x, self.y))

    def __repr__(self):
        return f"Point({self.x}, {self.y})"

# 使用
visited = set()
visited.add(Point(1, 2))
visited.add(Point(1, 2))  # 重复，不会添加
print(len(visited))  # 1
```

### 1. 自定义 Dataset 中的魔术方法

```python
from torch.utils.data import Dataset

class CustomImageDataset(Dataset):
    def __init__(self, images, labels, transform=None):
        self.images = images
        self.labels = labels
        self.transform = transform

    def __len__(self) -> int:
        return len(self.images)

    def __getitem__(self, index):
        from PIL import Image
        image = Image.open(self.images[index]).convert('RGB')
        if self.transform:
            image = self.transform(image)
        return image, self.labels[index]

    def __repr__(self):
        return f"CustomImageDataset(samples={len(self)}, classes={len(set(self.labels))})"
```

### 2. 批次数据类（支持解包和索引）

```python
import torch

class Batch:
    """批次数据，支持索引和属性访问"""

    def __init__(self, **kwargs):
        self._data = kwargs

    def __getitem__(self, key):
        return self._data[key]

    def __getattr__(self, name):
        if name.startswith('_'):
            raise AttributeError(name)
        try:
            return self._data[name]
        except KeyError:
            raise AttributeError(name)

    def __len__(self):
        for v in self._data.values():
            if hasattr(v, '__len__'):
                return len(v)
        return 0

    def __repr__(self):
        shapes = {k: getattr(v, 'shape', len(v)) for k, v in self._data.items()}
        return f"Batch(size={len(self)}, {shapes})"

    def to(self, device):
        for k, v in self._data.items():
            if hasattr(v, 'to'):
                self._data[k] = v.to(device)
        return self

# 使用
batch = Batch(
    images=torch.randn(32, 3, 224, 224),
    labels=torch.randint(0, 10, (32,))
)
print(batch['images'].shape)  # torch.Size([32, 3, 224, 224])
print(batch.images.shape)     # 同上
```

### 3. 实验结果类（支持排序和比较）

```python
from dataclasses import dataclass
from typing import Dict, Any

@dataclass
class ExperimentResult:
    """实验结果，支持排序"""
    name: str
    config: Dict[str, Any]
    metrics: Dict[str, float]

    def __lt__(self, other):
        return self.metrics.get('val_loss', float('inf')) < other.metrics.get('val_loss', float('inf'))

    def __repr__(self):
        return f"ExperimentResult(name={self.name!r}, val_loss={self.metrics.get('val_loss', 'N/A'):.4f})"

# 使用
results = [
    ExperimentResult("exp1", {}, {"val_loss": 0.5, "accuracy": 0.85}),
    ExperimentResult("exp2", {}, {"val_loss": 0.3, "accuracy": 0.90}),
]
best = min(results)  # 按 val_loss 排序
```

### 4. 上下文管理器用于计时

```python
import time
from contextlib import contextmanager

@contextmanager
def timer(name="Operation"):
    start = time.perf_counter()
    try:
        yield
    finally:
        end = time.perf_counter()
        print(f"{name} 耗时：{end - start:.4f} 秒")

# 使用
with timer("训练一个 epoch"):
    # 训练代码
    pass
```

## 完整魔术方法列表

| 类别 | 方法 | 用途 |
|------|------|------|
| 字符串 | `__str__`, `__repr__` | 字符串表示 |
| 数值 | `__add__`, `__sub__`, `__mul__` | 运算符重载 |
| 比较 | `__lt__`, `__eq__`, `__hash__` | 比较运算符 |
| 容器 | `__len__`, `__getitem__`, `__contains__` | 容器行为 |
| 迭代 | `__iter__`, `__next__` | 迭代器 |
| 上下文 | `__enter__`, `__exit__` | with 语句 |

## 性能提示

> **最佳实践**：
> - 始终实现 `__repr__`（便于调试）
> - 如果实现了 `__eq__`，也要实现 `__hash__`
> - 实现 `__len__` 和 `__getitem__` 可以让对象可迭代
