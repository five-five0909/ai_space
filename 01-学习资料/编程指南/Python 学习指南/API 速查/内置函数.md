## 字符与 ASCII 码转换

```python
# ord() - 字符 → ASCII 码（整数）
ord('a')    # 97
ord('A')    # 65
ord('0')    # 48
ord('中')   # 20013 (Unicode 码点)

# chr() - ASCII 码（整数）→ 字符
chr(97)     # 'a'
chr(65)     # 'A'
chr(48)     # '0'
chr(20013)  # '中'

# 常用 ASCII 码速查
# 'a'-'z': 97-122
# 'A'-'Z': 65-90
# '0'-'9': 48-57
```

> **提示**：`ord()` 实际返回的是 Unicode 码点，不仅限于 ASCII。

### LeetCode 实战：字符与 ASCII 码

```python
# LC 389: 找不同 - 利用 ASCII 码求和
def findTheDifference_sum(s: str, t: str) -> str:
    """t 比 s 多一个字符，求和相减"""
    return chr(sum(ord(c) for c in t) - sum(ord(c) for c in s))

# 字母偏移（凯撒密码）
def shift_char(c: str, offset: int) -> str:
    """字符偏移，如 'a' + 1 = 'b'"""
    return chr(ord(c) + offset)

# 判断大小写
def is_upper(c: str) -> bool:
    return 65 <= ord(c) <= 90

def is_lower(c: str) -> bool:
    return 97 <= ord(c) <= 122
```

### 除法运算符对比

```python
# 普通除法 / - 返回浮点数
7 / 3           # 2.3333333333333335
10 / 2          # 5.0 （注意：结果是浮点数）

# 整除 // - 向下取整，返回整数
7 // 3          # 2 （丢弃余数）
10 // 2         # 5 （整数结果）
-7 // 3         # -3 （向下取整，不是 -2！）

# 取余 % - 返回余数
7 % 3           # 1
10 % 2          # 0

# divmod() - 同时返回商和余数
divmod(7, 3)    # (2, 1) 相当于 (7 // 3, 7 % 3)
```

> **注意**：负数整除时向下取整！
> ```python
> -7 // 3        # -3（向下取整到更小的整数）
> -7 / 3        # -2.33...
> # 验证：(-7 // 3) * 3 + (-7 % 3) = -3 * 3 + 2 = -7 ✓
> ```

> **应用场景**：
> - 计算能分成多少份：`len(s) // len(sub)`
> - 数组索引计算：`idx // cols`（行号），`idx % cols`（列号）
> - 时间转换：`seconds // 3600`（小时），`seconds % 3600 // 60`（分钟）

## 位运算

### 基本运算符

```python
# 按位与 (&) - 同位都为1才为1
5 & 3    # 0101 & 0011 = 0001 = 1

# 按位或 (|) - 同位有1就为1
5 | 3    # 0101 | 0011 = 0111 = 7

# 按位异或 (^) - 同位不同为1，相同为0
5 ^ 3    # 0101 ^ 0011 = 0110 = 6

# 按位取反 (~)
~5       # -6（注意：Python 用补码表示负数）

# 左移 (<<) - 相当于乘 2^n
5 << 1   # 10 (5 * 2)
5 << 2   # 20 (5 * 4)

# 右移 (>>) - 相当于整除 2^n
20 >> 2  # 5 (20 // 4)
```

### 异或运算的关键性质

```python
# 性质 1：任何数和自己异或，结果为 0
a ^ a = 0
5 ^ 5    # 0

# 性质 2：任何数和 0 异或，结果为自己
a ^ 0 = a
5 ^ 0    # 5

# 性质 3：异或满足交换律和结合律
a ^ b ^ a = (a ^ a) ^ b = 0 ^ b = b

# 性质 4：连续异或可以消除成对出现的数
# 这个性质常用于"找唯一数"问题
```

> **提示**：异或运算 `^=` 是 `res = res ^ ord(c)` 的简写形式。

### LeetCode 实战：异或运算

```python
# LC 389: 找不同 - 异或法（推荐）
def findTheDifference_xor(s: str, t: str) -> str:
    """利用异或性质：成对抵消，剩下唯一"""
    res = 0
    for c in s + t:
        res ^= ord(c)    # 相同字符异或后归零
    return chr(res)      # 剩下的就是多出的字符

# LC 136: 只出现一次的数字
def singleNumber(nums: list) -> int:
    """所有数异或，成对的抵消，剩下的就是唯一数"""
    res = 0
    for num in nums:
        res ^= num
    return res

# LC 268: 丢失的数字
def missingNumber(nums: list) -> int:
    """找 0-n 中缺失的那个数"""
    n = len(nums)
    res = 0
    for i in range(n + 1):
        res ^= i        # 异或 0 到 n
    for num in nums:
        res ^= num      # 异或数组中的数
    return res          # 缺失的数会被留下
```

## collections 模块

### Counter 计数器

```python
from collections import Counter

# 基本用法：统计元素出现次数
Counter(['a', 'b', 'a', 'c', 'a', 'b'])
# Counter({'a': 3, 'b': 2, 'c': 1})

# 字符串作为可迭代对象
Counter("hello")
# Counter({'l': 2, 'h': 1, 'e': 1, 'o': 1})

# 常用方法
c = Counter("abracadabra")
c.most_common(2)    # [('a', 5), ('b', 2)] - 出现最多的 2 个
c['x']              # 0 - 不存在的 key 返回 0（不会报错）
c.update("abc")     # 增加计数
c.subtract("ab")    # 减少计数
```

### 字典相等比较

```python
# Counter 是 dict 的子类，支持相等比较
Counter("aab") == Counter("aba")   # True
Counter("aab") == Counter("abb")   # False

# 原理：比较所有 key-value 是否完全一致
{'a': 2, 'b': 1} == {'a': 2, 'b': 1}   # True
{'a': 2, 'b': 1} == {'a': 1, 'b': 2}   # False
```

> **提示**：普通字典比较顺序无关，只要 key-value 对相同即可。

### LeetCode 实战：Counter

```python
from collections import Counter

# LC 242: 有效的字母异位词
def isAnagram(s: str, t: str) -> bool:
    """两个字符串字母组成是否相同"""
    # Counter 直接统计每个字符出现次数，比较是否相等
    return Counter(s) == Counter(t)

# LC 387: 字符串中的第一个唯一字符
def firstUniqChar(s: str) -> int:
    """找第一个只出现一次的字符的索引"""
    count = Counter(s)
    for i, c in enumerate(s):
        if count[c] == 1:
            return i
    return -1

# LC 349: 两个数组的交集
def intersection(nums1: list, nums2: list) -> list:
    """找两个数组的公共元素（去重）"""
    return list(set(nums1) & set(nums2))
    # 或用 Counter：list((Counter(nums1) & Counter(nums2)).elements())
```

### 科研实战：Counter

```python
from collections import Counter

# 统计类别分布
labels = ['cat', 'dog', 'cat', 'bird', 'cat', 'dog']
distribution = Counter(labels)
# Counter({'cat': 3, 'dog': 2, 'bird': 1})

# 检查数据平衡
def is_balanced(labels: list, threshold: float = 0.1) -> bool:
    """检查类别是否平衡"""
    counts = Counter(labels).values()
    max_count, min_count = max(counts), min(counts)
    return (max_count - min_count) / max_count < threshold

# 统计 token 频率（NLP）
tokens = ["我", "爱", "Python", "我", "爱", "AI"]
token_freq = Counter(tokens)
# Counter({'我': 2, '爱': 2, 'Python': 1, 'AI': 1})
```

## LeetCode 实战场景

### 1. 两数之和 (LC 1)

```python
def two_sum(nums: list, target: int) -> list:
    seen = {}
    for i, num in enumerate(nums):
        complement = target - num
        if complement in seen:
            return [seen[complement], i]
        seen[num] = i
```

    min_v, max_v = min(values), max(values)
    return [(v - min_v) / (max_v - min_v) for v in values]
