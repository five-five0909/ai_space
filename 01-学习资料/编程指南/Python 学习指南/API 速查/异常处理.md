# 异常处理

## 基础语法

```python
try:
    result = 10 / 0
except ZeroDivisionError:
    print("不能除以零")
except Exception as e:
    print(f"其他错误：{e}")
else:
    print("没有异常时执行")
finally:
    print("总是执行")
```

## 主动抛出异常（raise）

`raise` 就是**主动抛出一个错误**，让程序停下来并告诉调用者"出问题了"。

### Python自动报错 vs 你主动报错

```python
# Python自己发现的错误
a = [1, 2, 3]
print(a[10])        # IndexError: list index out of range
                    # ↑ Python检测到越界，自动抛出

# 你主动抛出的错误
def broadcast_add(a, b):
    if not is_broadcastable(a.shape, b.shape):
        raise ValueError("形状不满足广播规则")  # 你决定在这里报错
        # ↑ raise之后，下面的代码全部不执行
    return a + b
```

### 常见错误类型

```python
raise ValueError("数值不合法")      # 值有问题，最常用
raise TypeError("类型不对")         # 类型有问题
raise IndexError("下标越界")        # 索引有问题
raise RuntimeError("运行时出错")    # 通用错误
raise KeyError("键不存在")          # 字典键不存在
```

### 捕获与不捕获的区别

```python
# 用 try/except 捕获，程序继续跑
try:
    result = broadcast_add(torch.ones(2), torch.ones(3, 4))
except ValueError as e:
    print(f"捕获到错误: {e}")
    result = None
print("程序没有崩溃，继续执行")  # ✓ 会执行

# 不捕获，程序直接崩
result = broadcast_add(torch.ones(2), torch.ones(3, 4))
# ValueError: 形状不满足广播规则: ...
# ↑ 程序在这行直接终止，后面全不执行
```

### 实战场景：广播规则验证器

```python
def check_broadcastable(shape_a, shape_b):
    """检查两个shape是否可以广播"""
    # 先补齐：短的左侧补1（模拟广播"右对齐"规则）
    max_len = max(len(shape_a), len(shape_b))
    shape_a = [1] * (max_len - len(shape_a)) + shape_a
    shape_b = [1] * (max_len - len(shape_b)) + shape_b

    # 逐维检查
    for dim_a, dim_b in zip(shape_a, shape_b):
        if dim_a != dim_b and dim_a != 1 and dim_b != 1:
            raise ValueError(
                f"形状不满足广播规则: dim_a={dim_a}, dim_b={dim_b} "
                f"（必须相等或其中一个为1）"
            )
    return True

# 为什么必须先补1再zip？
# 如果不补齐，zip会截断，漏掉最左边的维度！
shape_a = [2, 1]        # 2维
shape_b = [3, 1, 5]     # 3维

for dim_a, dim_b in zip(shape_a, shape_b):
    print(dim_a, dim_b)
# 输出:
# 2 3     ← 本来应该是 1 vs 3（a补1后第0维）
# 1 1     ← 本来应该是 2 vs 1
# 5 被完全忽略了！← 灾难

# 正确做法：先补1再zip
shape_a = [1] * (3 - 2) + [2, 1]  # [1, 2, 1]
shape_b = [3, 1, 5]               # 不变
for dim_a, dim_b in zip(shape_a, shape_b):
    print(dim_a, dim_b)
# 输出:
# 1 3     ← 正确：1可以拉伸成3 ✓
# 2 1     ← 正确：1可以拉伸成2 ✓
# 1 5     ← 正确：1可以拉伸成5 ✓
```

## LeetCode 实战场景

### 1. 安全的字典访问

```python
def get_value_safe(d: dict, key: str, default=None):
    try:
        return d[key]
    except KeyError:
        return default

# 或使用 get 方法
value = d.get(key, default)
```

### 2. 整数转换

```python
def parse_int(s: str, default: int = 0) -> int:
    try:
        return int(s)
    except (ValueError, TypeError):
        return default
```

## 科研实战场景

### 1. 模型加载容错

```python
import torch

def load_model_safe(model_path: str, default_model=None):
    try:
        return torch.load(model_path)
    except FileNotFoundError:
        print(f"模型文件不存在：{model_path}")
        return default_model
    except RuntimeError as e:
        print(f"模型加载失败：{e}")
        return default_model
```

### 2. 实验重试机制

```python
import time

def train_with_retry(train_fn, max_retries=3):
    for attempt in range(max_retries):
        try:
            return train_fn()
        except RuntimeError as e:
            if attempt == max_retries - 1:
                raise
            print(f"训练失败，{attempt + 1}/{max_retries}, 重试中...")
            time.sleep(2 ** attempt)  # 指数退避
```

## 上下文管理器

```python
# 使用 with 语句
with open('file.txt', 'r') as f:
    content = f.read()
# 自动关闭文件

# 自定义上下文管理器 - 计时器
from contextlib import contextmanager
import time

@contextmanager
def timer(name: str):
    start = time.time()
    try:
        yield
    finally:
        end = time.time()
        print(f"{name} 耗时：{end - start:.2f}秒")

with timer("训练"):
    train()
```

## 最佳实践

```python
# 捕获具体异常
try:
    int(user_input)
except ValueError:
    print("请输入有效数字")

# 避免过度捕获
try:
    do_something()
except Exception as e:  # 不要使用空 except
    log_error(e)
    raise

# 使用 finally 清理资源
f = open('file.txt')
try:
    process(f)
finally:
    f.close()
```
