│
├── 02-AI研究/            # AI研究与知识库
│   ├── ai_knowledge_base/ # AI技术知识库（43个主题目录）
│   │   ├── 01-mamba/     # Mamba架构
│   │   ├── 02-transformer/ # Transformer架构
│   │   ├── 03-attention/  # 注意力机制
│   │   ├── ...           # 更多主题
│   │   └── 43-pisfm/     # 主题索引与分类

├── 03-项目开发/          # 开发项目
│   ├── prompt-space/     # Prompt工程与技能模块

├── 04-工具与文档/        # 工具与文档
